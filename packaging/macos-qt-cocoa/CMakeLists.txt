cmake_minimum_required(VERSION 3.21)
project(QtCocoaOwnershipFix VERSION 6.12.0 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Gui CorePrivate GuiPrivate)
# Quote the official SDK's -idirafter path when the external volume has spaces.
get_target_property(platform_options Qt6::Platform INTERFACE_COMPILE_OPTIONS)
string(REPLACE "SHELL:-idirafter " "-idirafter;" platform_options "${platform_options}")
set_target_properties(Qt6::Platform PROPERTIES INTERFACE_COMPILE_OPTIONS "${platform_options}")
if(NOT QTBASE_SOURCE_DIR)
    message(FATAL_ERROR "Set QTBASE_SOURCE_DIR to the patched Qt 6.12.0 source directory")
endif()
set(COCOA_SOURCE "${QTBASE_SOURCE_DIR}/src/plugins/platforms/cocoa")
file(GLOB COCOA_SOURCES "${COCOA_SOURCE}/*.mm" "${COCOA_SOURCE}/*.h")
list(FILTER COCOA_SOURCES EXCLUDE REGEX "/qnsview_[^/]+\\.mm$")
list(FILTER COCOA_SOURCES EXCLUDE REGEX "/qcocoavulkaninstance\\.(mm|h)$")
list(APPEND COCOA_SOURCES "${COCOA_SOURCE}/qcocoasessionmanager.cpp")
qt_add_plugin(QCocoaIntegrationPlugin SHARED PLUGIN_TYPE platforms CLASS_NAME QCocoaIntegrationPlugin)
target_sources(QCocoaIntegrationPlugin PRIVATE ${COCOA_SOURCES})
target_include_directories(QCocoaIntegrationPlugin PRIVATE "${COCOA_SOURCE}")
target_link_libraries(QCocoaIntegrationPlugin PRIVATE Qt6::Core Qt6::CorePrivate Qt6::Gui Qt6::GuiPrivate)
foreach(fw Foundation AppKit Carbon CoreServices CoreVideo IOKit IOSurface Metal QuartzCore UniformTypeIdentifiers ScreenCaptureKit)
    find_library(FW${fw} ${fw} REQUIRED)
    target_link_libraries(QCocoaIntegrationPlugin PRIVATE ${FW${fw}})
endforeach()
target_compile_options(QCocoaIntegrationPlugin PRIVATE -fno-exceptions)
target_compile_definitions(QCocoaIntegrationPlugin PRIVATE QT_NO_EXCEPTIONS)
set_target_properties(QCocoaIntegrationPlugin PROPERTIES OUTPUT_NAME qcocoa INSTALL_RPATH "@loader_path/../../lib")
qt_add_resources(QCocoaIntegrationPlugin qcocoaresources PREFIX "/qt-project.org/mac/cursors" BASE "${COCOA_SOURCE}" FILES "${COCOA_SOURCE}/images/sizeallcursor.png" "${COCOA_SOURCE}/images/spincursor.png" "${COCOA_SOURCE}/images/waitcursor.png")
